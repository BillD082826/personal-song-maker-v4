<script>

  function escapeHtml(value) {

    return String(value ?? "")

      .replaceAll("&", "&amp;")

      .replaceAll("<", "&lt;")

      .replaceAll(">", "&gt;")

      .replaceAll('"', "&quot;")

      .replaceAll("'", "&#039;");

  }


  function formatDate(value) {

    if (!value) return "";

    return new Date(value).toLocaleString();

  }


  async function loadOrders() {

    const ordersContainer = document.getElementById("orders");

    const orderCount = document.getElementById("orderCount");


    ordersContainer.innerHTML =

      '<div class="loading">Loading orders...</div>';


    try {

      const response = await fetch("/api/admin/orders");


      if (!response.ok) {

        throw new Error("Could not load orders.");

      }


      const data = await response.json();

      const orders = Array.isArray(data) ? data : data.orders || [];


      orderCount.textContent = orders.length;


      if (orders.length === 0) {

        ordersContainer.innerHTML =

          '<div class="empty">No customer orders yet.</div>';

        return;

      }


      ordersContainer.innerHTML = `

        <div class="orders">

          ${orders.map(order => {


            const statusClass =

              String(order.status).toLowerCase() === "paid"

                ? "status-paid"

                : "status-new";


            return `

              <div class="order-card">


                <div class="order-header">

                  <div class="order-number">

                    ${escapeHtml(order.id)}

                  </div>


                  <div class="status ${statusClass}">

                    ${escapeHtml(order.status)}

                  </div>

                </div>


                <div class="details">


                  <div class="detail">

                    <span class="label">Customer</span>

                    ${escapeHtml(order.customer_name)}

                  </div>


                  <div class="detail">

                    <span class="label">Email</span>

                    ${escapeHtml(order.email)}

                  </div>


                  <div class="detail">

                    <span class="label">Song For</span>

                    ${escapeHtml(order.person)}

                  </div>


                  <div class="detail">

                    <span class="label">Occasion</span>

                    ${escapeHtml(order.occasion)}

                  </div>


                  <div class="detail">

                    <span class="label">Music Style</span>

                    ${escapeHtml(order.style)}

                  </div>


                  <div class="detail">

                    <span class="label">Mood</span>

                    ${escapeHtml(order.mood)}

                  </div>


                  <div class="detail">

                    <span class="label">Ordered</span>

                    ${escapeHtml(formatDate(order.created_at))}

                  </div>


                </div>


                <div class="story">

                  <span class="label">Their Story</span>

                  ${escapeHtml(order.story)}

                </div>


                ${

                  order.message

                    ? `

                      <div class="message">

                        <span class="label">Special Message</span>

                        ${escapeHtml(order.message)}

                      </div>

                    `

                    : ""

                }


              </div>

            `;

          }).join("")}

        </div>

      `;


    } catch (error) {

      console.error(error);


      orderCount.textContent = "—";


      ordersContainer.innerHTML =

        '<div class="error">Could not load customer orders.</div>';

    }

  }


  loadOrders();

</script>
